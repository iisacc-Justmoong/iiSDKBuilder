cmake_minimum_required(VERSION 3.31)
project(iiSDKBuilderConsumer LANGUAGES CXX)
find_package(iiSDKBuilder 0.4 CONFIG REQUIRED)
if(NOT COMMAND iisdk_builder_cpython_aot OR NOT EXISTS "${iiSDKBuilder_DIR}/CompileAot.py")
    message(FATAL_ERROR "The installed Builder must expose its CPython AOT helper and backend")
endif()
find_package(iiSDKCore 0.20 CONFIG REQUIRED COMPONENTS Embedded)
include("${iiSDKBuilder_DIR}/iiSDKBuilderMatrix.cmake")
add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE iiSDKBuilder::iiSDKBuilder iiSDKCore::Embedded)
enable_testing()
add_test(NAME consumer COMMAND consumer)
