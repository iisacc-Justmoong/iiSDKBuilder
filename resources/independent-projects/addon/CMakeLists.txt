cmake_minimum_required(VERSION 3.31)
project(IndependentAddonFixture LANGUAGES CXX)
find_package(iiSDKCore CONFIG QUIET)
if(iiSDKCore_FOUND)
    message(FATAL_ERROR "An independent SDK must not discover the selected Core prefix")
endif()
unset(AMBIENT_CORE_HEADER CACHE)
find_path(AMBIENT_CORE_HEADER NAMES iiSDKLeakProbe.h)
if(AMBIENT_CORE_HEADER)
    message(FATAL_ERROR "An independent SDK must not discover a Core header through the ambient CMake include path")
endif()
include(CheckIncludeFileCXX)
unset(AMBIENT_CORE_COMPILER_HEADER CACHE)
check_include_file_cxx(iiSDKLeakProbe.h AMBIENT_CORE_COMPILER_HEADER)
if(AMBIENT_CORE_COMPILER_HEADER)
    message(FATAL_ERROR "An independent SDK must not compile a Core header through ambient compiler paths")
endif()
if(TEST_EXPLICIT_INCLUDE)
    unset(EXPLICIT_EXTERNAL_HEADER CACHE)
    find_path(EXPLICIT_EXTERNAL_HEADER NAMES ExternalDependency.h)
    if(NOT EXPLICIT_EXTERNAL_HEADER)
        message(FATAL_ERROR "A kit's explicit external include path must be available")
    endif()
endif()
add_library(FixtureAddon INTERFACE)
set_target_properties(FixtureAddon PROPERTIES EXPORT_NAME Addon)
install(TARGETS FixtureAddon EXPORT AddonTargets)
install(EXPORT AddonTargets NAMESPACE Addon:: DESTINATION lib/cmake/Addon)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/AddonConfig.cmake"
    "include(\"\${CMAKE_CURRENT_LIST_DIR}/AddonTargets.cmake\")\n")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/AddonConfig.cmake" DESTINATION lib/cmake/Addon)
