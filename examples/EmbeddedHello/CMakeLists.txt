cmake_minimum_required(VERSION 3.31)
project(EmbeddedHello LANGUAGES CXX)
find_package(iiSDKCore CONFIG REQUIRED COMPONENTS Embedded)
add_executable(EmbeddedHello platform/main.cpp)
target_link_libraries(EmbeddedHello PRIVATE iiSDKCore::Embedded)
target_compile_features(EmbeddedHello PRIVATE cxx_std_23)
option(IISDK_EXAMPLE_CORTEX_M3 "Use the explicit LM3S6965 Cortex-M3 BSP" OFF)
if(IISDK_EXAMPLE_CORTEX_M3)
    target_sources(EmbeddedHello PRIVATE platform/cortex_m3.cpp)
    target_compile_definitions(EmbeddedHello PRIVATE IISDK_EXAMPLE_CORTEX_M3=1)
    target_compile_options(EmbeddedHello PRIVATE -fno-builtin)
    target_link_options(EmbeddedHello PRIVATE "-T${CMAKE_CURRENT_SOURCE_DIR}/platform/cortex_m3.ld")
    set_target_properties(EmbeddedHello PROPERTIES SUFFIX .elf)
endif()
