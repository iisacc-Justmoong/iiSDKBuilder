cmake_minimum_required(VERSION 3.31)
project(iiSDKBuilderHello LANGUAGES CXX)
find_package(iiSDKCore 0.15 CONFIG REQUIRED COMPONENTS Runtime)
if(ANDROID)
    add_library(iiSDKBuilderHello SHARED main.cpp)
else()
    add_executable(iiSDKBuilderHello main.cpp)
endif()
target_link_libraries(iiSDKBuilderHello PRIVATE iiSDKCore::Runtime)
set_target_properties(iiSDKBuilderHello PROPERTIES CXX_EXTENSIONS OFF)
if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
    set_target_properties(iiSDKBuilderHello PROPERTIES MACOSX_BUNDLE TRUE
        MACOSX_BUNDLE_GUI_IDENTIFIER "com.iisacc.iisdkbuilder.hello")
endif()

include(CTest)
if(BUILD_TESTING AND NOT CMAKE_CROSSCOMPILING)
    add_test(NAME iiSDKBuilder.Hello COMMAND iiSDKBuilderHello)
    set_tests_properties(iiSDKBuilder.Hello PROPERTIES PASS_REGULAR_EXPRESSION "Hello, World!")
endif()
