cmake_minimum_required(VERSION 3.31)
project(HelloResources VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(iiSDKCore 0.17 CONFIG REQUIRED COMPONENTS Runtime)
if(NOT IISDKBUILDER_RESOURCES_MODULE)
    set(IISDKBUILDER_RESOURCES_MODULE "${CMAKE_CURRENT_LIST_DIR}/../../cmake/iiSDKBuilderResources.cmake")
endif()
include("${IISDKBUILDER_RESOURCES_MODULE}")
if(ANDROID)
    add_library(HelloResources SHARED main.cpp)
else()
    add_executable(HelloResources main.cpp)
endif()
target_link_libraries(HelloResources PRIVATE iiSDKCore::Runtime)
iisdkbuilder_add_resources(TARGET HelloResources MANIFEST resources.json)
if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
    set_target_properties(HelloResources PROPERTIES MACOSX_BUNDLE TRUE
        MACOSX_BUNDLE_GUI_IDENTIFIER com.iisacc.hello-resources)
endif()
include(CTest)
if(BUILD_TESTING AND NOT CMAKE_CROSSCOMPILING)
    add_test(NAME HelloResources.native COMMAND HelloResources)
    set_tests_properties(HelloResources.native PROPERTIES
        PASS_REGULAR_EXPRESSION "NATIVE_RESOURCES_OK.*embedded=true.*deduplicated=true")
endif()
